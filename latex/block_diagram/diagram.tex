\documentclass[border=10pt]{standalone}
\usepackage[utf8]{inputenc}
\usepackage[T1]{fontenc}
\usepackage{amsmath}
\usepackage{tikz}
\usetikzlibrary{shapes.geometric, arrows.meta, positioning, fit, backgrounds, calc}

% Definizione palette colori
\definecolor{primary}{HTML}{2B4C7E}
\definecolor{accent}{HTML}{C05A46}
\definecolor{boxbg}{HTML}{F4F6F9}
\definecolor{innerbg}{HTML}{FFFFFF}
\definecolor{bordercolor}{HTML}{333333}

\begin{document}

\begin{tikzpicture}[
    auto,
    >=Stealth,
    node distance=1.0cm and 1.2cm,
    % Stili
    subbox/.style={
        draw=bordercolor,
        fill=innerbg,
        thick,
        rounded corners=8pt,
        align=left,
        inner sep=8pt,
        minimum width=4.8cm,
        font=\small
    },
    subsubbox/.style={
        draw=bordercolor!70,
        fill=boxbg!60,
        thick,
        rounded corners=4pt,
        align=left,
        inner sep=6pt,
        font=\footnotesize
    },
    ovalbox/.style={
        draw=bordercolor,
        fill=innerbg,
        thick,
        ellipse,
        align=center,
        inner sep=6pt,
        minimum width=2.6cm,
        minimum height=1.1cm,
        font=\small\bfseries
    },
    labelstyle/.style={
        font=\footnotesize\bfseries,
        text=primary,
        fill=white,
        inner sep=2pt,
        rounded corners=2pt
    }
]

% ===================================================================
% 1. BLOCCO PRINCIPALE: SIMULATE
% ===================================================================

\node [subbox] (source) {
    \textbf{\large Source}\\[4pt]
    \textbullet~spectrum \\
    \textbullet~beam shape
};

\node [subbox, below=of source] (sensor) {
    \textbf{\large Sensor}\\[4pt]
    \textbullet~material \\
    \textbullet~thickness \\
    \textbullet~diff. coeff. $D$
};

\node [subbox, below=of sensor] (readout) {
    \textbf{\large Readout}\\[4pt]
    \textbullet~chip layout\\[6pt]
    \begin{tikzpicture}[baseline=(sub.north)]
        \node[subsubbox] (sub) {
            \textbullet~ENC \\
            \textbullet~pedestal \\
            \textbullet~gain
        };
    \end{tikzpicture}
};

% Frecce interne simulate
\draw [->, thick] (source) -- node[labelstyle, right=2pt] {$(E, x, y)$} (sensor);
\draw [->, thick] (sensor) -- node[labelstyle, right=2pt] {$(E, x, y, z)$} (readout);

% Contenitore esterno simulate
\begin{scope}[on background layer]
    \node [
        draw=primary,
        fill=boxbg,
        ultra thick,
        fit=(source) (sensor) (readout),
        inner sep=14pt,
        yshift=-8pt
    ] (simulate) {};
    \node[anchor=north west, primary, font=\bfseries\Large] at (simulate.north west) {simulate};
\end{scope}

% ===================================================================
% 2. BLOCCO PRINCIPALE: CONVERT
% ===================================================================
\node [
    draw=primary,
    fill=boxbg,
    ultra thick,
    minimum width=3.8cm,
    minimum height=1.6cm,
    font=\bfseries\Large,
    text=primary,
    anchor=south west
] (convert) at ($(simulate.south east) + (4.0,0)$) {convert};

% ===================================================================
% 3. BLOCCO PRINCIPALE: CALIBGEN (con nodi ovali interni)
% ===================================================================

% Disposizione multilivello dei nodi ovali
\node [ovalbox, below=3.2cm of simulate.south, xshift=-1.0cm] (noise) {noise};
\node [ovalbox, right=1.5cm of noise] (pedestal) {pedestal};
\node [ovalbox, below=1.8cm of noise, xshift=1.2cm] (equalization) {equalization};
\node [ovalbox, below=1.8cm of equalization] (position) {position};

% Ovale speciale synthesize
\node [ovalbox, right=1.2cm of pedestal] (synthesize) {synthesize};

% Contenitore esterno calibgen
\begin{scope}[on background layer]
    \node [
        draw=primary,
        fill=boxbg,
        ultra thick,
        fit=(noise) (pedestal) (equalization) (position) (synthesize),
        inner sep=16pt,
        yshift=-6pt
    ] (calibgen) {};
    \node[anchor=north west, primary, font=\bfseries\Large] at (calibgen.north west) {calibgen};
\end{scope}

% ===================================================================
% 4. COLLEGAMENTI E FRECCE
% ===================================================================

% Punto di unione per l'ingresso in calibgen
\coordinate (merge) at ($(calibgen.north) + (-1.0cm, 1.2cm)$);

% Percorso da simulate a merge
\draw [thick] (readout.south) -- (simulate.south |- simulate.south) 
      -- node[labelstyle, left=2pt] {$(E, x, y, z, \text{PHA})$} (simulate.south |- merge)
      -- (merge);

% Percorso da convert a merge
\draw [thick] (convert.south) |- (merge);

% Freccia principale .hdf verso calibgen
\draw [thick, primary] (merge) -- node[labelstyle, right=4pt] {.hdf} (calibgen.north |- merge) -- (calibgen.north |- noise.north);

% Ramificazione dell'ingresso .hdf ai 4 blocchi ovali
\draw [->, thick, primary] (calibgen.north |- noise.north) -- (noise.north);
\draw [->, thick, primary] (calibgen.north |- noise.north) -| (pedestal.north);
\draw [->, thick, primary] (calibgen.north |- noise.north) -| (equalization.north);
\draw [->, thick, primary] (calibgen.north |- noise.north) -| (position.north);

% Frecce interne tra blocchi ovali in calibgen
% Noise -> Equalization & Position
\draw [->, thick] (noise.south) -- node[labelstyle, left=1pt] {.hdf} (equalization.north west);
\draw [->, thick] (noise.south) .. controls ($(noise.south)+(0,-2.5)$) and ($(position.west)+(-1.0,0)$) .. node[labelstyle, left=1pt] {.hdf} (position.west);

% Pedestal -> Equalization & Position
\draw [->, thick] (pedestal.south) -- node[labelstyle, right=1pt] {.hdf} (equalization.north east);
\draw [->, thick] (pedestal.south) .. controls ($(pedestal.south)+(0,-2.5)$) and ($(position.east)+(1.0,0)$) .. node[labelstyle, right=1pt] {.hdf} (position.east);

% Equalization -> Position
\draw [->, thick] (equalization.south) -- (position.north);

% Freccia da Synthesize verso Readout (sottoblocco ENC, pedestal, gain)
\draw [->, thick, accent, dashed] (synthesize.north) .. controls ($(synthesize.north)+(0,2.5)$) and ($(readout.east)+(3.0,0)$) .. node[labelstyle, text=accent, above right] {sintesi parametri} (readout.east);

\end{tikzpicture}

\end{document}
